\section{Задание}
\begin{bbttask}
  \item реализовать наивный поиск всех вхождений образца в текст со счётчиком сравнений
  символов;\label{task:naive}
  \item реализовать вычисление префикс-функции и поиск по алгоритму Кнута — Морриса — Пратта
  (КМП) со счётчиком сравнений;\label{task:kmp}
  \item сгенерировать по seed варианта (100 + номер варианта) два класса входов — случайный
  текст над алфавитом из четырёх букв и худший для наивного поиска случай — для пяти длин
  текста с шагом 2, длина образца — $n/20$; выполнить замеры: прогрев, пять повторов,
  медиана;\label{task:bench}
  \item построить график времени от длины текста в логарифмических осях, сопоставить рост
  времени с оценками $O(n \cdot m)$ и $O(n + m)$, объяснить разницу между классами
  входов;\label{task:analysis}
  \item проверить корректность: сверка с поиском встроенными средствами языка на нескольких
  сотнях случайных входов и граничные случаи — пустой текст, образец длиннее текста,
  перекрывающиеся вхождения.\label{task:check}
\end{bbttask}
