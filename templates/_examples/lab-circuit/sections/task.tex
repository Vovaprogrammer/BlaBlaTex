\section{Задание}
\begin{bbttask}
  \item собрать на стенде последовательную RC-цепь по варианту, включить амперметр и
  вольтметр на конденсаторе, привести схему;\label{task:scheme}
  \item выполнить предварительный расчёт: сопротивление конденсатора, полное сопротивление,
  ток, напряжение на конденсаторе и сдвиг фаз для всех частот;\label{task:calc}
  \item измерить ток и напряжение на конденсаторе на тех же частотах;\label{task:measure}
  \item свести расчёт и эксперимент в таблицу, вычислить расхождение в процентах, построить
  графики $I(f)$ и $U_C(f)$;\label{task:compare}
  \item объяснить характер зависимостей и причины расхождения расчёта и
  эксперимента.\label{task:explain}
\end{bbttask}
