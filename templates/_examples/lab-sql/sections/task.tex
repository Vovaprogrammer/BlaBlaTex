\section{Задание}
\begin{bbttask}
  \item создать в SQLite базу данных библиотеки из трёх таблиц — читатели, книги, выдачи — с
  первичными и внешними ключами и заполнить её тестовыми данными, привести схему базы
  данных;\label{task:db}
  \item составить запрос с условием и сортировкой: книги одного автора, новые
  сначала;\label{task:where}
  \item составить запрос с соединением таблиц: книги, которые сейчас на руках, с именами
  читателей и сроками возврата;\label{task:join}
  \item составить запрос с группировкой: сколько раз каждый читатель брал книги, только
  читатели с несколькими выдачами;\label{task:group}
  \item составить запрос с вычислением: просроченные выдачи на дату проверки и число дней
  просрочки;\label{task:late}
  \item составить вложенный запрос: книги, которые ни разу не выдавались.\label{task:sub}
\end{bbttask}
